cmake_minimum_required(VERSION 3.5)
project(pm_recording)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl REQUIRED)
find_package(fastrtps REQUIRED)
# Foxyのrmw_fastrtps exportが参照するmessage targetを先に定義する。
find_package(rmw_dds_common REQUIRED)
find_package(rmw_fastrtps_shared_cpp REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(rosbag2_transport REQUIRED)
find_package(rosbag2_storage REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
add_executable(async_recorder src/async_recorder.cpp)
target_include_directories(async_recorder PRIVATE include)
ament_target_dependencies(async_recorder rclcpp rosbag2_cpp rosbag2_transport rosbag2_storage)
target_link_libraries(async_recorder yaml-cpp Threads::Threads)
install(TARGETS async_recorder DESTINATION lib/${PROJECT_NAME})
add_executable(timing_recorder src/async_recorder.cpp)
target_compile_definitions(timing_recorder PRIVATE PM_TIMING_RECORDER)
target_include_directories(timing_recorder PRIVATE include)
ament_target_dependencies(timing_recorder rclcpp rosbag2_cpp rosbag2_transport rosbag2_storage)
target_link_libraries(timing_recorder yaml-cpp Threads::Threads)
install(TARGETS timing_recorder DESTINATION lib/${PROJECT_NAME})
# 明示的なLD_PRELOAD試験だけで使う診断。通常recorderへはリンクしない。
add_library(rcl_receive_trace SHARED src/rcl_receive_trace.cpp)
ament_target_dependencies(rcl_receive_trace rcl fastrtps rmw_fastrtps_shared_cpp)
target_link_libraries(rcl_receive_trace fastrtps ${CMAKE_DL_LIBS})
install(TARGETS rcl_receive_trace DESTINATION lib)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_async_writer test/test_async_writer.cpp)
  target_include_directories(test_async_writer PRIVATE include)
  ament_target_dependencies(test_async_writer rosbag2_cpp rosbag2_storage)
  target_link_libraries(test_async_writer Threads::Threads)
endif()
ament_package()
